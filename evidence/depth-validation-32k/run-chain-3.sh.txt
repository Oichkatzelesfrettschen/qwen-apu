#!/bin/sh
# Fill and decode every admitted row that lacks a 32768 tuple, one arm each at
# the registry geometry, then the review model through the router with its
# projector loaded. Outputs land inside this worktree.
set -u
tree=$HOME/<checkout>
out=$HOME/<worktree>/evidence/depth-validation-32k
export QWEN_WEDGE_DEPTHS=32768 QWEN_WEDGE_GEOMETRIES=128:32 QWEN_WEDGE_CONDITIONAL_DEPTHS=32768
for pair in \
  qwen38-2b-uncensored:Qwen3.8-2B-Uncensored-GGUF/Qwen3.8-2B-Uncensored-Q4_K_M.gguf \
  qwen35-2b-unredacted:Qwen3.5-2B-Unredacted-MAX-GGUF/Qwen3.5-2B-Unredacted-MAX.Q4_K_M.gguf \
  qwen35-2b-heretic:Qwen3.5-2B-Opus-Distilled-Heretic-GGUF/Qwen3.5-2B-Opus-Distilled-Heretic-Thinking-Multistage-SFT-v1.0.Q4_K_M.gguf \
  qwen35-2b-hauhau:Qwen3.5-2B-Uncensored-HauhauCS-Aggressive-GGUF/Qwen3.5-2B-Uncensored-HauhauCS-Aggressive-Q4_K_M.gguf \
  qwen35-08b-unsloth-unc:Qwen3.5-0.8B-Uncensored-GGUF/Qwen3.5-0.8B.Q4_K_M.gguf \
  qwenseer-2b:Qwenseer-2B-GGUF/Qwenseer-2B.Q4_K_M.gguf; do
  id=${pair%%:*}; rel=${pair#*:}
  printf "chain_start id=%s utc=%s\n" "$id" "$(date -u +%FT%TZ)"
  "$tree/remote/probe-depth-wedge.sh" "$tree/.runtime/models/$rel" "$out/$id" > "$out/$id.driver.log" 2>&1
  printf "chain_stop id=%s status=%s utc=%s\n" "$id" "$?" "$(date -u +%FT%TZ)"
done
printf "chain_start id=lfm25-vl-16b-router-child utc=%s\n" "$(date -u +%FT%TZ)"
"$tree/remote/probe-depth-projector.sh" --runtime-mode router-child lfm25-vl-16b "$out/lfm25-vl-16b-router-child" > "$out/lfm25-vl-16b-router-child.driver.log" 2>&1
printf "chain_stop id=lfm25-vl-16b-router-child status=%s utc=%s\n" "$?" "$(date -u +%FT%TZ)"
printf "chain_done utc=%s\n" "$(date -u +%FT%TZ)"
